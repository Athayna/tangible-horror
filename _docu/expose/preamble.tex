\usepackage[

  sorting=none,                 % Sort by citation order
  backend=biber,                % Use biber backend for better Unicode support
  urldate=long,                 % Full date format for URL access dates

  giveninits=false,             % Show full first names

\addbibresource{bibliography.bib}

\DeclareFieldFormat{title}{\mkbibquote{\textit{#1}}}   % Format titles in italics with quotes
\DeclareFieldFormat{url}{\url{#1}}                     % Format URLs
\renewcommand*{\mkbibnamegiven}[1]{#1}                 % Full first names
